```sql
-- Migration: Create investigation_notes table for analyst notes on entities
-- Entity IDs reference Neo4j internal IDs (not PostgreSQL entities table UUIDs)
-- because the entity detail page uses Neo4j IDs in URLs

CREATE TABLE IF NOT EXISTS investigation_notes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    entity_id TEXT NOT NULL,  -- Neo4j entity ID (string), not FK to entities table
    content TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_investigation_notes_entity_id ON investigation_notes(entity_id);
CREATE INDEX IF NOT EXISTS idx_investigation_notes_created_at ON investigation_notes(created_at DESC);

-- Add biography columns to entities table for AI-generated biographies
ALTER TABLE entities ADD COLUMN IF NOT EXISTS biography TEXT;
ALTER TABLE entities ADD COLUMN IF NOT EXISTS biography_generated_at TIMESTAMPTZ;
ALTER TABLE entities ADD COLUMN IF NOT EXISTS biography_model TEXT;
```

**GET /api/entities/[id]/notes** — List all notes for entity
```typescript
import { json } from '@sveltejs/kit';
import type { RequestHandler } from './$types';
import { query } from '$lib/server/db';

export const GET: RequestHandler = async ({ params, platform }) => {
  if (!platform?.env) return json({ error: 'Platform unavailable' }, { status: 500 });

  const { id } = params;
  if (!id || !/^\d+$/.test(id)) return json({ error: 'Invalid entity ID' }, { status: 400 });

  const notes = await query(platform,
    'SELECT id, entity_id, content, created_at, updated_at FROM investigation_notes WHERE entity_id = $1 ORDER BY created_at DESC',
    [id]
  );
  return json(notes);
};
```

**POST /api/entities/[id]/notes** — Create note
```typescript
export const POST: RequestHandler = async ({ params, request, platform }) => {
  if (!platform?.env) return json({ error: 'Platform unavailable' }, { status: 500 });

  const { id } = params;
  if (!id || !/^\d+$/.test(id)) return json({ error: 'Invalid entity ID' }, { status: 400 });

  const body = await request.json();
  const content = (body as { content?: string }).content?.trim();
  if (!content) return json({ error: 'Content is required' }, { status: 400 });
  if (content.length > 10000) return json({ error: 'Content too long (max 10000 chars)' }, { status: 400 });

  const notes = await query(platform,
    'INSERT INTO investigation_notes (entity_id, content) VALUES ($1, $2) RETURNING id, entity_id, content, created_at, updated_at',
    [id, content]
  );
  return json(notes[0], { status: 201 });
};
```

**PUT /api/entities/[id]/notes** — Update note (note_id in body)
```typescript
export const PUT: RequestHandler = async ({ params, request, platform }) => {
  if (!platform?.env) return json({ error: 'Platform unavailable' }, { status: 500 });

  const { id } = params;
  if (!id || !/^\d+$/.test(id)) return json({ error: 'Invalid entity ID' }, { status: 400 });

  const body = await request.json();
  const { note_id, content } = body as { note_id?: string; content?: string };
  if (!note_id) return json({ error: 'note_id is required' }, { status: 400 });
  const trimmed = content?.trim();
  if (!trimmed) return json({ error: 'Content is required' }, { status: 400 });
  if (trimmed.length > 10000) return json({ error: 'Content too long' }, { status: 400 });

  const notes = await query(platform,
    'UPDATE investigation_notes SET content = $1, updated_at = NOW() WHERE id = $2 AND entity_id = $3 RETURNING id, entity_id, content, created_at, updated_at',
    [trimmed, note_id, id]
  );
  if (notes.length === 0) return json({ error: 'Note not found' }, { status: 404 });
  return json(notes[0]);
};
```

**DELETE /api/entities/[id]/notes** — Delete note (note_id in body)
```typescript
export const DELETE: RequestHandler = async ({ params, request, platform }) => {
  if (!platform?.env) return json({ error: 'Platform unavailable' }, { status: 500 });

  const { id } = params;
  if (!id || !/^\d+$/.test(id)) return json({ error: 'Invalid entity ID' }, { status: 400 });

  const body = await request.json();
  const { note_id } = body as { note_id?: string };
  if (!note_id) return json({ error: 'note_id is required' }, { status: 400 });

  const result = await query(platform,
    'DELETE FROM investigation_notes WHERE id = $1 AND entity_id = $2 RETURNING id',
    [note_id, id]
  );
  if (result.length === 0) return json({ error: 'Note not found' }, { status: 404 });
  return json({ deleted: true });
};
```

All endpoints validate entity ID format (numeric only, matching Neo4j internal IDs). All SQL uses parameterized queries (SEC-01 pattern).

```typescript
async function createNote(content: string) {
  saving = true;
  try {
    const res = await fetch(`/api/entities/${entityId}/notes`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ content }),
    });
    if (!res.ok) throw new Error('Failed to create note');
    const note = await res.json();
    notes = [note, ...notes];
    isAdding = false;
  } finally { saving = false; }
}
```
