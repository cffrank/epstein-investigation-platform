**Generate function:**
```typescript
async function generateBiography() {
  isGenerating = true;
  streamedContent = '';
  toolCalls = [];
  citations = [];
  error = null;

  try {
    const response = await fetch(`/api/entities/${entityId}/biography`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name: entityName }),
    });

    if (!response.ok || !response.body) throw new Error('Biography generation failed');

    // Reuse parseSSE from chat
    for await (const event of parseSSE(response.body)) {
      // Handle same events as chat store: tool_call, tool_result, text_delta, citations_delta, done, error
      switch (event.event) {
        case 'text_delta': {
          const data = JSON.parse(event.data);
          streamedContent += data.text;
          break;
        }
        case 'tool_call': {
          const data = JSON.parse(event.data);
          toolCalls = [...toolCalls, { id: data.id, name: data.name, status: 'running' }];
          break;
        }
        case 'tool_result': {
          const data = JSON.parse(event.data);
          const tc = toolCalls.find(t => t.id === data.id);
          if (tc) { tc.status = data.status || 'complete'; tc.resultCount = data.resultCount; }
          toolCalls = [...toolCalls];
          break;
        }
        case 'citations_delta': {
          const data = JSON.parse(event.data);
          citations = [...citations, data.citation];
          break;
        }
        case 'done':
          currentBio = {
            content: streamedContent,
            generated_at: new Date().toISOString(),
            model: 'claude-sonnet-4-6',
            citations,
          };
          break;
        case 'error': {
          const data = JSON.parse(event.data);
          error = data.message;
          break;
        }
      }
    }
  } catch (e) {
    error = e instanceof Error ? e.message : 'Unknown error';
  } finally {
    isGenerating = false;
  }
}
```
